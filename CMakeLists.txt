cmake_minimum_required(VERSION 3.10)
include(CMakeDefaults.cmake)
project(APath VERSION 1.0.0 LANGUAGES C)
include(CTest)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wstrict-prototypes)
    if(APATH_WARNINGS_AS_ERRORS)
        add_compile_options(-Werror)
    endif()
    if(APATH_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -fsanitize=address,undefined")
    endif()
endif()

add_library(apath_core STATIC src/core/ap_map.c src/core/ap_history.c src/core/ap_nav.c)
target_include_directories(apath_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/core")
if(NOT MSVC)
    target_link_libraries(apath_core PUBLIC m)
endif()

if(NOT APATH_CORE_ONLY)
    if(NOT MINGW)
        message(FATAL_ERROR "The editor requires MinGW. On Windows use a MinGW shell and -G \"MinGW Makefiles\", or use a supplied cross toolchain.")
    endif()
    enable_language(RC)
    add_library(apath_ui STATIC src/app.c src/win_files.c)
    target_include_directories(apath_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_compile_definitions(apath_ui PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0601)
    target_link_libraries(apath_ui PUBLIC apath_core user32 gdi32 comdlg32 comctl32 shell32 advapi32)
    add_executable(apath WIN32 src/main.c src/apath.rc)
    target_include_directories(apath PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(apath PRIVATE apath_ui)
    set_property(TARGET apath APPEND_STRING PROPERTY LINK_FLAGS " -municode -static-libgcc")
    set_property(SOURCE src/apath.rc APPEND PROPERTY OBJECT_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/src/apath.manifest" "${CMAKE_CURRENT_SOURCE_DIR}/res/apath.ico")
    install(TARGETS apath RUNTIME DESTINATION .)
endif()

if(APATH_BUILD_TESTS)
    add_executable(apath_core_tests tests/test_core.c)
    target_link_libraries(apath_core_tests PRIVATE apath_core)
    add_test(NAME core COMMAND apath_core_tests)
    if(NOT APATH_CORE_ONLY)
        add_executable(apath_ui_tests tests/test_ui.c src/apath.rc)
        target_include_directories(apath_ui_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
        target_link_libraries(apath_ui_tests PRIVATE apath_ui)
        set_property(TARGET apath_ui_tests APPEND_STRING PROPERTY LINK_FLAGS " -static-libgcc")
        add_test(NAME windows_ui COMMAND apath_ui_tests)
    endif()
endif()
